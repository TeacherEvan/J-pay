1. Executive Triage Summary

J-pay has evolved through rapid multi-phase architectural pivots—most notably transitioning from a pure Convex backend (convex/*.ts) to a custom RPC/Adapter layer (src/lib/backend/runtime.ts, src/lib/backend/db.ts, src/app/api/backend/call/route.ts) backed by Supabase PostgreSQL (ADR-004-supabase-authoritative-backend.md) and multi-tenant organizations (ADR-005-multi-org-per-deployment.md). While the test harness in scripts/test-*.mjs (34 custom MJS test runners) demonstrates strong engineering intent around shift derivation, idempotency, and org isolation, the repository carries severe dual-stack split-brain debt: legacy convex/*.ts modules coexist alongside src/lib/backend/* adapters, and 62 duplicated skill files sit in both .agents/skills/convex-* and .claude/skills/convex-*. Primary production risks center on RPC authorization boundary enforcement in /api/backend/call, offline outbox replay ordering during shift/travel transitions, and client-side polling/deduplication loops in src/lib/backend/react.ts.

Refactor 1: Monotonic Generation Counter in React Backend Adapter (src/lib/backend/react.ts)

    Target Files: src/lib/backend/react.ts   
    teacherevan/j-pay

    Why It Matters: Prevents a slow in-flight /api/backend/call query response from overwriting fresher state when a mutation invalidates the cache while the query is still pending over high-latency mobile networks.   
    teacherevan/j-pay

    Blueprint:

        Before (Conceptual):
        TypeScript

        // In-flight fetch resolves and blindly updates cache/state
        const data = await fetchBackendCall(fnName, args);
        queryCache.set(cacheKey, data);
        setState({ data, error: null, isLoading: false });

        After (Conceptual):
        TypeScript

        // Stamp every fetch with the current mutation generation counter
        const fetchGen = globalMutationGeneration;
        const data = await fetchBackendCall(fnName, args);
        if (fetchGen < globalMutationGeneration) {
          // A mutation occurred while this query was in flight; discard stale payload
          return;
        }
        queryCache.set(cacheKey, { data, gen: fetchGen });
        setState({ data, error: null, isLoading: false });

Refactor 2: Unified Mutex Lock Between Outbox Flush and Live Check-In (src/lib/outbox.ts & src/app/check-in/live-submit.tsx)

    Target Files: src/lib/outbox.ts, src/app/use-outbox.ts, src/app/check-in/live-submit.tsx   
    teacherevan/j-pay

    Why It Matters: Eliminates race conditions where background outbox flushing and manual button presses submit overlapping punch/travel transitions simultaneously.   
    teacherevan/j-pay

    Blueprint:

        Before (Conceptual):
        TypeScript

        // use-outbox.ts flushes on 'online' independently of live-submit.tsx busy latch
        window.addEventListener("online", () => flushOutbox(submitPunch));

        After (Conceptual):
        TypeScript

        // Shared async lock serializes all outbox flushes and live check-in submissions
        let syncPromise: Promise<void> | null = null;
        export async function withPunchSyncLock<T>(fn: () => Promise<T>): Promise<T> {
          while (syncPromise) await syncPromise.catch(() => {});
          let release!: () => void;
          syncPromise = new Promise<void>((r) => { release = r; });
          try {
            return await fn();
          } finally {
            syncPromise = null;
            release();
          }
        }

// TARGET PATTERN for src/lib/backend/db.ts
const TENANT_SCOPED_TABLES = new Set([
  "users", "sites", "logs", "shifts", "travel_logs",
  "shift_overrides", "messages", "manager_sites"
]);

export class ScopedSupabaseDb {
  constructor(
    private readonly client: SupabaseClient,
    private readonly callerOrgId: string | null,
    private readonly isSuperModerator: boolean = false
  ) {}

  private enforceTenantFilter<T extends { eq: (col: string, val: string) => T }>(
    table: string,
    queryBuilder: T
  ): T {
    if (TENANT_SCOPED_TABLES.has(table) && this.callerOrgId && !this.isSuperModerator) {
      return queryBuilder.eq("org_id", this.callerOrgId);
    }
    return queryBuilder;
  }

  async patch(table: string, id: string, values: Record<string, unknown>) {
    // Prevents Org A from ever patching an ID owned by Org B
    let q = this.client.from(table).update(values).eq("id", id);
    q = this.enforceTenantFilter(table, q);
    const { data, error } = await q.select().single();
    if (error || !data) throw new Error(`UNAUTHORIZED_OR_MISSING_RECORD: ${table}:${id}`);
    return data;
  }
}

-- TARGET PATTERN: supabase/migrations/0019_atomic_punch_rpc.sql
CREATE OR REPLACE FUNCTION public.commit_punch_and_upsert_shift(
  p_org_id UUID,
  p_user_id UUID,
  p_site_id UUID,
  p_idempotency_key TEXT,
  p_punch_type TEXT,
  p_punched_at TIMESTAMPTZ,
  p_lat DOUBLE PRECISION,
  p_lng DOUBLE PRECISION,
  p_shift_started_at TIMESTAMPTZ,
  p_shift_ended_at TIMESTAMPTZ,
  p_duration_minutes INTEGER
) RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_log_id UUID;
  v_shift_id UUID;
BEGIN
  -- 1. Idempotent log insert (0002_logs_idempotency_unique.sql)
  INSERT INTO public.logs (org_id, user_id, site_id, idempotency_key, type, timestamp, lat, lng)
  VALUES (p_org_id, p_user_id, p_site_id, p_idempotency_key, p_punch_type, p_punched_at, p_lat, p_lng)
  ON CONFLICT (user_id, idempotency_key) DO UPDATE SET idempotency_key = EXCLUDED.idempotency_key
  RETURNING id INTO v_log_id;

  -- 2. Atomic shift upsert within the exact same database transaction (0015_shifts_unique_user_started.sql)
  IF p_shift_started_at IS NOT NULL THEN
    INSERT INTO public.shifts (org_id, user_id, site_id, started_at, ended_at, duration_minutes)
    VALUES (p_org_id, p_user_id, p_site_id, p_shift_started_at, p_shift_ended_at, p_duration_minutes)
    ON CONFLICT (user_id, started_at)
    DO UPDATE SET
      ended_at = COALESCE(EXCLUDED.ended_at, public.shifts.ended_at),
      duration_minutes = COALESCE(EXCLUDED.duration_minutes, public.shifts.duration_minutes),
      updated_at = NOW()
    RETURNING id INTO v_shift_id;
  END IF;

  RETURN jsonb_build_object('log_id', v_log_id, 'shift_id', v_shift_id);
END;
$$;

REVOKE ALL ON FUNCTION public.commit_punch_and_upsert_shift FROM PUBLIC, anon, authenticated;

// TARGET PATTERN for src/lib/csv.ts
const FORMULA_TRIGGER_RE = /^[=+\-@\t\r]/;

export function escapeCsvCell(raw: unknown): string {
  if (raw === null || raw === undefined) return "";
  let str = String(raw);
  if (FORMULA_TRIGGER_RE.test(str)) {
    str = `'${str}`; // Neutralize spreadsheet formula evaluation
  }
  if (/[",\r\n]/.test(str)) {
    return `"${str.replace(/"/g, '""')}"`;
  }
  return str;
}
